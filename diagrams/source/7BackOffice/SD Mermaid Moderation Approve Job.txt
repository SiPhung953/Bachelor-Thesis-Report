sequenceDiagram
    actor BackOfficeWorker as Back-Office Worker
    participant FrontendUI as «boundary»<br/>Frontend UI
    participant JwtAuthentication as «security»<br/>JWTAuthentication
    participant JobModerationController as «control»<br/>JobModerationController
    participant JobModerationService as «service»<br/>JobModerationService
    participant PrismaClient as «persistence»<br/>PrismaClient
    participant AuditLogger as «utility»<br/>AuditLogger
    participant Database as «database»<br/>Database

    BackOfficeWorker->>+FrontendUI: 1 : Select job posting and click Approve
    FrontendUI->>+JwtAuthentication: 2 : approveJob(accessToken, jobId)

    JwtAuthentication->>JwtAuthentication: 3 : verifyAccessToken(accessToken)

    %% GUARD 1: Authentication Break
    break User is not authenticated
        JwtAuthentication-->>FrontendUI: 4a : HTTP 401 Unauthorized
        FrontendUI-->>BackOfficeWorker: 5a : Redirect to Login page
    end

    JwtAuthentication->>JwtAuthentication: 4 : verifyScopes(ADMIN)

    %% GUARD 2: Token Scope Break
    break Token scope does not include ADMIN
        JwtAuthentication-->>FrontendUI: 5b : HTTP 403 Forbidden
        FrontendUI-->>BackOfficeWorker: 6b : Display role restriction warning
    end

    JwtAuthentication->>+JobModerationController: 5 : approveJob(jobId)
    deactivate JwtAuthentication

    JobModerationController->>+JobModerationService: 6 : approveJob(jobId, currentUser)

    JobModerationService->>JobModerationService: 7 : assertAdmin(currentUser)

    %% GUARD 3: Service-Level Role Break (defence in depth)
    break User does not have Admin role
        JobModerationService-->>JobModerationController: 8a : ForbiddenRoleError
        JobModerationController-->>FrontendUI: 9a : HTTP 403 Forbidden
        FrontendUI-->>BackOfficeWorker: 10a : Display role restriction warning
    end

    %% Retrieve Job Posting
    JobModerationService->>+PrismaClient: 8 : findJobPostingById(jobId)
    PrismaClient->>+Database: 9 : fetchJobPosting(jobId)
    Database-->>-PrismaClient: 10 : jobData
    PrismaClient-->>-JobModerationService: 11 : job

    %% GUARD 4: Job Existence Break
    break Job Posting not found
        JobModerationService-->>JobModerationController: 12a : JobNotFoundError
        JobModerationController-->>FrontendUI: 13a : HTTP 404 Not Found
        FrontendUI-->>BackOfficeWorker: 14a : Display job not found message
    end

    %% GUARD 5: Idempotency Break
    break Job Posting is already ACTIVE
        JobModerationService-->>JobModerationController: 12b : InvalidJobStatusError
        JobModerationController-->>FrontendUI: 13b : HTTP 409 Conflict
        FrontendUI-->>BackOfficeWorker: 14b : Display already approved message
    end

    %% GUARD 6: Illegal State Transition Break
    break Job Posting is not PENDING_APPROVAL
        JobModerationService-->>JobModerationController: 12c : InvalidJobStatusError
        JobModerationController-->>FrontendUI: 13c : HTTP 400 Bad Request
        FrontendUI-->>BackOfficeWorker: 14c : Display invalid job status message
    end

    %% GUARD 7: Deadline Break
    break Deadline has already passed
        JobModerationService-->>JobModerationController: 12d : InvalidDeadlineError
        JobModerationController-->>FrontendUI: 13d : HTTP 400 Bad Request
        FrontendUI-->>BackOfficeWorker: 14d : Display expired deadline message
    end

    %% Approve Job Posting and record the audit trail atomically
    rect rgb(240, 240, 240)
        note over JobModerationService, Database: Single transaction — an approval without<br/>its audit record is rolled back
        JobModerationService->>+PrismaClient: 12 : updateJobStatus(jobId, ACTIVE, approvedAt)
        PrismaClient->>+Database: 13 : update jobs
        Database-->>-PrismaClient: 14 : updatedJob
        PrismaClient-->>-JobModerationService: 15 : updatedJob

        JobModerationService->>+AuditLogger: 16 : log(JOB_APPROVED, jobId, currentUser)
        AuditLogger->>+PrismaClient: 17 : createLog()
        PrismaClient->>+Database: 18 : insert audit log
        Database-->>-PrismaClient: 19 : logRecord
        PrismaClient-->>-AuditLogger: 20 : logRecord
        AuditLogger-->>-JobModerationService: 21 : void
    end

    JobModerationService-->>-JobModerationController: 22 : ApproveJobResponse
    JobModerationController-->>-FrontendUI: 23 : HTTP 200 OK + ApproveJobResponse
    FrontendUI-->>-BackOfficeWorker: 24 : Display approval success message
