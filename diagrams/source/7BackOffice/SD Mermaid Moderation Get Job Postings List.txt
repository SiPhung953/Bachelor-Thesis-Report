sequenceDiagram
    actor BackOfficeWorker as Back-Office Worker
    participant FrontendUI as «boundary»<br/>Frontend UI
    participant JwtAuthentication as «security»<br/>JWTAuthentication
    participant JobModerationController as «control»<br/>JobModerationController
    participant JobModerationService as «service»<br/>JobModerationService
    participant PrismaClient as «persistence»<br/>PrismaClient
    participant Database as «database»<br/>Database

    BackOfficeWorker->>+FrontendUI: 1 : Open job moderation page
    FrontendUI->>+JwtAuthentication: 2 : getJobs(accessToken)

    JwtAuthentication->>JwtAuthentication: 3 : verifyAccessToken(accessToken)

    %% GUARD 1: Authentication Break
    break User is not authenticated
        JwtAuthentication-->>FrontendUI: 4a : HTTP 401 Unauthorized
        FrontendUI-->>BackOfficeWorker: 5a : Redirect to Login page
    end

    JwtAuthentication->>JwtAuthentication: 4 : assertAdmin()

    %% GUARD 2: Role Authorization Break
    break User does not have Admin role
        JwtAuthentication-->>FrontendUI: 5b : HTTP 403 Forbidden
        FrontendUI-->>BackOfficeWorker: 6b : Display role restriction warning
    end

    JwtAuthentication->>+JobModerationController: 5 : getJobs()
    deactivate JwtAuthentication

    JobModerationController->>+JobModerationService: 6 : getJobs()

    %% Retrieve Job Postings
    JobModerationService->>+PrismaClient: 7 : findManyJobPostings()
    PrismaClient->>+Database: 8 : fetchJobPostings()
    Database-->>-PrismaClient: 9 : jobData[]
    PrismaClient-->>-JobModerationService: 10 : jobs[]

    %% Empty Result or not, the flow goes the same
    JobModerationService-->>-JobModerationController: 11 : JobResponse[]
    JobModerationController-->>FrontendUI: 12 : HTTP 200 OK + JobResponse[]
    alt response is empty
	FrontendUI-->>BackOfficeWorker: 13a : Display empty state
    else response is not empty
    	FrontendUI-->>BackOfficeWorker: 13b : Display job posting list
    end
