  }

  class JobPosting {
    <<Entity>>
    +UUID id
    +String title
    +UUID companyId
    +JobStatus status [PENDING_APPROVAL|ACTIVE|REJECTED|CLOSED|EXPIRED|DELETED]
  }

  class Log {
    <<Entity>>
    +UUID id
    +UUID actorId
    +LogAction action
    +LogTargetType targetType [USER|JOB]
    +UUID targetId
    +LogLevel level [USER, JOB, APPLICATION, RESUME, COMPANY, AUTH, SYSTEM]
    +String message
    +DateTime createdAt
  }
