sequenceDiagram
    actor BackOfficeWorker as Back-Office Worker
    participant FrontendUI as «boundary»<br/>Frontend UI
    participant JwtAuthentication as «security»<br/>JWTAuthentication
    participant JobModerationController as «control»<br/>JobModerationController
    participant JobModerationService as «service»<br/>JobModerationService
    participant PrismaClient as «persistence»<br/>PrismaClient
    participant AuditLogger as «utility»<br/>AuditLogger
    participant Database as «database»<br/>Database

    BackOfficeWorker->>+FrontendUI: 1 : Select job posting and click Reject
    FrontendUI->>+JwtAuthentication: 2 : rejectJob(accessToken, jobId, rejectionReason)

    JwtAuthentication->>JwtAuthentication: 3 : verifyAccessToken(accessToken)

    %% GUARD 1: Authentication Break
    break User is not authenticated
        JwtAuthentication-->>FrontendUI: 4a : HTTP 401 Unauthorized
        FrontendUI-->>BackOfficeWorker: 5a : Redirect to Login page
    end

    JwtAuthentication->>JwtAuthentication: 4 : assertAdmin()

    %% GUARD 2: Role Authorization Break
    break User does not have Admin role
        JwtAuthentication-->>FrontendUI: 5b : HTTP 403 Forbidden
        FrontendUI-->>BackOfficeWorker: 6b : Display role restriction warning
    end

    JwtAuthentication->>+JobModerationController: 5 : rejectJob(jobId, rejectionReason)
    deactivate JwtAuthentication

    JobModerationController->>+JobModerationService: 6 : rejectJob(jobId, rejectionReason, currentUser)

    %% Retrieve Job Posting
    JobModerationService->>+PrismaClient: 7 : findJobPostingById(jobId)
    PrismaClient->>+Database: 8 : fetchJobPosting(jobId)
    Database-->>-PrismaClient: 9 : jobData
    PrismaClient-->>-JobModerationService: 10 : job

    %% GUARD 3: Job Existence Break
    break Job Posting not found
        JobModerationService-->>JobModerationController: 11a : JobNotFoundError
        JobModerationController-->>FrontendUI: 12a : HTTP 404 Not Found
        FrontendUI-->>BackOfficeWorker: 13a : Display job not found message
    end

    break Job Posting is already REJECTED
	JobModerationService-->>JobModerationController: 11b : InvalidJobStatusError
        JobModerationController-->>FrontendUI: 12b : HTTP 409 Conflict
        FrontendUI-->>BackOfficeWorker: 13b : Display already rejected message
    end

    %% GUARD 4: Job Posting non-rejectable status break
    break Job Posting is not PENDING_APPROVAL
        JobModerationService-->>JobModerationController: 11c : InvalidJobStatusError
        JobModerationController-->>FrontendUI: 12c : HTTP 400 Bad Request
        FrontendUI-->>BackOfficeWorker: 13c : Display invalid job status message
    end

    %% Reject Job Posting
    JobModerationService->>+PrismaClient: 11c : updateJobStatus(jobId, REJECTED, rejectedAt, rejectionReason)
    PrismaClient->>+Database: 12c : update jobs
    Database-->>-PrismaClient: 13c : updatedJob
    PrismaClient-->>-JobModerationService: 14c : updatedJob

    %% Audit Log
    JobModerationService->>+AuditLogger: 15c : log(JOB_REJECTED, jobId, currentUser)
    AuditLogger->>+PrismaClient: 16c : createLog()
    PrismaClient->>+Database: 17c : insert audit log
    Database-->>-PrismaClient: 18c : logRecord
    PrismaClient-->>-AuditLogger: 19c : logRecord
    AuditLogger-->>-JobModerationService: 20c : void

    JobModerationService-->>-JobModerationController: 21c : JobResponse
    JobModerationController-->>-FrontendUI: 22c : HTTP 200 OK + JobResponse
    FrontendUI-->>-BackOfficeWorker: 23c : Display rejection success message
