  %% ==========================================

  class AuthController {
    <<Controller>>
    +register(RegisterRequest) RegisterResponse
    +login(LoginRequest) LoginResponse
    +logout() LogoutResponse
    +requestPasswordReset(ForgotPasswordRequest) ForgotPasswordResponse
    +resetPassword(ResetPasswordRequest) ResetPasswordResponse
    +validateResetToken(token) Boolean
  }

  %% ==========================================
  %% 2. APPLICATION LAYER (Services)
  %% ==========================================
  class AuthService {
    <<Service>>
    +register(RegisterRequest) RegisterResponse
    +login(LoginRequest) LoginResponse
    +logout() LogoutResponse
    +requestPasswordReset(ForgotPasswordRequest) ForgotPasswordResponse
    +resetPassword(ResetPasswordRequest) ResetPasswordResponse
    +validateResetToken(token) boolean
  }

  %% ==========================================
  %% 3. INFRASTRUCTURE & UTILITY LAYER
  %% ==========================================
  class PasswordHasher {
    <<Utility>>
    +hash(password) String
    +compare(password, paswordHash) boolean
  }

  class ResetTokenUtils {
    <<Utility>>
    +generateToken() String
    +hashToken(token) String
  }

  class EmailService {
    <<Utility>>
    +sendPasswordResetEmail(email, resetUrl) void
  }

  class PrismaClient {
    <<Utility>>
    +user
    +passwordResetToken
    +role
  }

  %% ==========================================
  %% 4. DOMAIN LAYER (Entities)
  %% ==========================================
  class User {
    <<Entity>>
    +UUID id
    +String email
    +String passwordHash
    +UserStatus status [ACTIVE|BANNED]
    +DateTime createdAt
    +DateTime updatedAt
  }

  class PasswordResetToken {
    <<Entity>>
    +UUID id
    +UUID userId
    +String tokenHash
    +DateTime expiresAt
    +DateTime usedAt

    +Int id
    +String name
  }
