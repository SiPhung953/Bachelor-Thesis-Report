sequenceDiagram
    actor Guest as : Guest
    participant FrontendUI as «boundary»<br/>: Frontend UI
    participant AuthController as «control»<br/>: AuthController
    participant AuthService as «service»<br/>: AuthService
    participant EmailService as «service»<br/>: EmailService
    participant PrismaClient as «persistence»<br/>: PrismaClient
    participant ResetTokenUtils as «utility»<br/>: ResetTokenUtils
    participant Database as «database»<br/>: Database

        Guest->>+FrontendUI: 1 : User go to Login page and<br/>press "Forgot Password".<br/>Then type in email
        FrontendUI->>+AuthController: 2 : requestPasswordReset(email)
        AuthController->>+AuthService: 3 : requestPasswordReset(email)
        
        AuthService->>+PrismaClient: 4 : findUserByEmail(email)
        PrismaClient-->>-AuthService: 5 : user
		
		alt Condition: User doesn't exist
			AuthService->>AuthService: 6 : Do not generate token or<br/>send email
      AuthService-->>AuthController: 7 : successResponse
      AuthController-->>FrontendUI: 8 : HTTP 200 ok
      FrontendUI-->>Guest: 9 : Display generic success response
        
    else Condition: User exist
      AuthService->>+ResetTokenUtils: 6 : generateResetToken()
      ResetTokenUtils-->>-AuthService: 7 : token
            
      AuthService->>PrismaClient: 8 : saveResetToken(token)
      activate PrismaClient
      PrismaClient->>Database: 9 : insertToken()
      activate Database
      Database-->>PrismaClient: 10 : created record
      deactivate Database
      deactivate PrismaClient

      AuthService->>AuthService: 11 : generateResetUrl(token)
      AuthService->>+EmailService: 12 : sendResetEmail(email, url)
      EmailService-->>-AuthService: 13 : email sent
			
      AuthService-->>-AuthController: 15 : successResponse
      AuthController-->>-FrontendUI: 16 : HTTP 200 ok
      FrontendUI-->>-Guest: 17 : Display generic success response
    end
