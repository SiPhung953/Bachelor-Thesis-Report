sequenceDiagram
    actor AuthenticatedUser as : Authenticated User
    participant FrontendUI as «boundary»<br/>: Frontend UI
    participant JwtAuthentication as «security»<br/>: JWTAuthentication
    participant ProfileController as «control»<br/>: ProfileController
    participant ProfileService as «service»<br/>: ProfileService
    participant PasswordHasher as «utility»<br/>: PasswordHasher
    participant PrismaClient as «persistence»<br/>: PrismaClient
    participant Database as «database»<br/>: Database

    AuthenticatedUser->>FrontendUI: 1 : Enter current password and new password
    AuthenticatedUser->>+FrontendUI: 2 : Click Change Password

    FrontendUI->>JwtAuthentication: 3 : changePassword(passwordPayload)

    activate JwtAuthentication
    JwtAuthentication->>JwtAuthentication: 4 : verifyAccessToken(accessToken)

    alt Condition: User is not authenticated
        JwtAuthentication-->>FrontendUI: 5 : HTTP 401 Unauthorized
        FrontendUI-->>AuthenticatedUser: 6 : Redirect to Login page

    else Condition: User is authenticated
        JwtAuthentication->>ProfileController: 5 : changePassword(user, passwordPayload)
        deactivate JwtAuthentication
        activate ProfileController

        ProfileController->>ProfileService: 6 : changePassword(userId, passwordPayload)
        activate ProfileService

        ProfileService->>+PrismaClient: 7 : findUserById(userId)
        PrismaClient->>+Database: 8 : fetchPasswordHash(userId)
        Database-->>-PrismaClient: 9 : userData
        PrismaClient-->>-ProfileService: 10 : user

        ProfileService->>+PasswordHasher: 11 : compare(currentPassword, storedHash)
        PasswordHasher-->>-ProfileService: 12 : comparisonResult

        alt Condition: Current password is incorrect
            ProfileService-->>ProfileController: 13a : invalidCurrentPasswordError
            ProfileController-->>FrontendUI: 14a : HTTP 400 Bad Request
            FrontendUI-->>AuthenticatedUser: 15a : Display password error

        else Condition: Current password is correct
            
            alt Condition: New password equals current password
                ProfileService-->>ProfileController: 13b : identicalPasswordError
                ProfileController-->>FrontendUI: 14b : HTTP 400 Bad Request
                FrontendUI-->>AuthenticatedUser: 15b : Display "New password must be different" message
            
            else Condition: New password is different
                ProfileService->>+PasswordHasher: 13 : hash(newPassword)
                PasswordHasher-->>-ProfileService: 14 : newPasswordHash

                ProfileService->>+PrismaClient: 15 : updatePassword(userId, newPasswordHash)
                PrismaClient->>+Database: 16 : saveNewPasswordHash(userId, newPasswordHash)
                Database-->>-PrismaClient: 17 : updatedUserData
                PrismaClient-->>-ProfileService: 18 : updatedUser

                ProfileService-->>-ProfileController: 19 : ChangePasswordResponse

                ProfileController-->>-FrontendUI: 20 : HTTP 200 OK<br/>ChangePasswordResponse

                FrontendUI-->>-AuthenticatedUser: 21 : Display success message
            end
        end
    end
